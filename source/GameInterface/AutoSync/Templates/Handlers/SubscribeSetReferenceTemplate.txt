Subscribe<{{MemberType}}, {{ MemberDeclaringTypeName}}_{{ MemberName }}_SetLocalMessage>((instanceId, data) => 
{
    int index = 0;
    uint valueId = 0;
    if (data.Value != null)
    {
        index = AutoSync.Constants.TypeMap[data.Value.GetType()];
        if (!this.objectManager.TryGetHandle(data.Value, out valueId)) return;
    }
                
    network.SendAll(new {{ MemberDeclaringTypeName}}_{{ MemberName }}_SetNetworkMessage(instanceId, valueId, index));
});

SubscribeNetwork<{{MemberType}}, {{ MemberDeclaringTypeName}}_{{ MemberName }}_SetNetworkMessage>((instance, data) => 
{
    object value = null;
    if (data.ValueId != 0)
    {
        var type = AutoSync.Constants.TypeMapReverse[data.ValueTypeIndex];
        if (!objectManager.TryGetObjectWithLogging(data.ValueId, out value)) return;
        if (!type.IsInstanceOfType(value)) return;
    }
{{~ if ReadOnly ~}}
	AccessTools.Field(typeof({{MemberDeclaringType}}), "{{MemberName}}").SetValue(instance, value);
{{~ else ~}}
	using (new AllowedThread())
	{
		instance.{{MemberName}} = ({{MemberType}}) value;
	}
{{~ end ~}}
});
